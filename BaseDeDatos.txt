CREATE DATABASE DBEcoPuntos;
go

-- Usar la base de datos DBEcoPuntos
USE DBEcoPuntos;
GO

IF NOT EXISTS (SELECT * FROM sys.schemas WHERE name = 'dbo')
BEGIN
    EXEC('CREATE SCHEMA [dbo]');
END;

CREATE TABLE [dbo].[Usuario] (
    [Id] int NOT NULL IDENTITY(1,1),
    [Nombre] nvarchar(100) NOT NULL,
    [Cedula] nvarchar(100) NOT NULL UNIQUE,
    [Correo] nvarchar(255) NOT NULL UNIQUE,
    [Direccion] nvarchar(255),
    [Telefono] nvarchar(20),
    [Contrasena] nvarchar(255),
    PRIMARY KEY ([Id])
);


CREATE TABLE [dbo].[Empleado] (
    [Id] int NOT NULL,
    [CodigoDeEmpleado] nvarchar(50) NOT NULL UNIQUE,
    PRIMARY KEY ([Id])
);


CREATE TABLE [dbo].[Cliente] (
    [Id] int NOT NULL,
    [EcoPuntos] int DEFAULT 0,
    PRIMARY KEY ([Id])
);


CREATE TABLE [dbo].[Trabajador] (
    [Id] int NOT NULL,
    [Horario] nvarchar(100) NOT NULL,
    PRIMARY KEY ([Id])
);


CREATE TABLE [dbo].[EcoPunto] (
    [Id] int NOT NULL IDENTITY(1,1),
    [IdTrabajador] int NOT NULL,
    [Latitud] nvarchar(255),
    [Longitud] nvarchar(255),
    [Direccion] nvarchar(255) NOT NULL,
    [Horario] nvarchar(100) NOT NULL,
    [NombreEcopunto] nvarchar(255) NOT NULL,
    PRIMARY KEY ([Id])
);


CREATE TABLE [dbo].[Material] (
    [Id] int NOT NULL IDENTITY(1,1),
    [Nombre] nvarchar(150) NOT NULL,
    CONSTRAINT [pk_Material_Id] PRIMARY KEY ([Id])
);

-- Indexes
CREATE UNIQUE INDEX [dbo_pk_Material_Id]
ON [dbo].[Material] ([Id]);


CREATE TABLE [dbo].[MaterialEntrega] (
    [IdEntrega] int NOT NULL,
    [IdMaterial] int NOT NULL,
    [Peso] float NOT NULL,
    [Puntos] int NOT NULL,
    [Estado] bit NOT NULL,
    CONSTRAINT [pk_MaterialEntrega] PRIMARY KEY ([IdEntrega], [IdMaterial])
);


CREATE TABLE [dbo].[Entrega] (
    [Id] int NOT NULL IDENTITY(1,1),
    [IdCliente] int NOT NULL,
    [IdTrabajador] int,
    [IdEcopunto] int,
    CONSTRAINT [pk_Entrega_id] PRIMARY KEY ([Id])
);

-- Indexes
CREATE UNIQUE INDEX [dbo_pk_Entrega_id]
ON [dbo].[Entrega] ([Id]);


CREATE TABLE [dbo].[MaterialEcoPunto] (
    [IdEcoPunto] int NOT NULL,
    [IdMaterial] int NOT NULL,
    PRIMARY KEY ([IdEcoPunto], [IdMaterial])
);

-- Tabla Administrador
CREATE TABLE [dbo].[Administrador] (
    Id INT PRIMARY KEY,
    NivelAcceso INT NOT NULL DEFAULT 1
);


-- Foreign key constraints CORREGIDAS
-- Schema: dbo
ALTER TABLE [dbo].[Cliente] ADD CONSTRAINT [FK_Cliente_Id] FOREIGN KEY([Id]) REFERENCES [dbo].[Usuario]([Id]);
ALTER TABLE [dbo].[Empleado] ADD CONSTRAINT [FK_Empleado_Id] FOREIGN KEY([Id]) REFERENCES [dbo].[Usuario]([Id]);
ALTER TABLE [dbo].[Trabajador] ADD CONSTRAINT [FK_Trabajador_Id] FOREIGN KEY([Id]) REFERENCES [dbo].[Empleado]([Id]);
ALTER TABLE [dbo].[Administrador] ADD CONSTRAINT [FK_Administrador_Id] FOREIGN KEY([Id]) REFERENCES [dbo].[Empleado]([Id]);

ALTER TABLE [dbo].[EcoPunto] ADD CONSTRAINT [FK_EcoPunto_IdTrabajador] FOREIGN KEY([IdTrabajador]) REFERENCES [dbo].[Trabajador]([Id]);

ALTER TABLE [dbo].[Entrega] ADD CONSTRAINT [FK_Entrega_IdCliente] FOREIGN KEY([IdCliente]) REFERENCES [dbo].[Cliente]([Id]);
ALTER TABLE [dbo].[Entrega] ADD CONSTRAINT [FK_Entrega_IdTrabajador] FOREIGN KEY([IdTrabajador]) REFERENCES [dbo].[Trabajador]([Id]);
ALTER TABLE [dbo].[Entrega] ADD CONSTRAINT [FK_Entrega_IdEcopunto] FOREIGN KEY([IdEcopunto]) REFERENCES [dbo].[EcoPunto]([Id]);

ALTER TABLE [dbo].[MaterialEntrega] ADD CONSTRAINT [FK_MaterialEntrega_IdEntrega] FOREIGN KEY([IdEntrega]) REFERENCES [dbo].[Entrega]([Id]);
ALTER TABLE [dbo].[MaterialEntrega] ADD CONSTRAINT [FK_MaterialEntrega_IdMaterial] FOREIGN KEY([IdMaterial]) REFERENCES [dbo].[Material]([Id]);

ALTER TABLE [dbo].[MaterialEcoPunto] ADD CONSTRAINT [FK_MaterialEcoPunto_IdEcoPunto] FOREIGN KEY([IdEcoPunto]) REFERENCES [dbo].[EcoPunto]([Id]);
ALTER TABLE [dbo].[MaterialEcoPunto] ADD CONSTRAINT [FK_MaterialEcoPunto_IdMaterial] FOREIGN KEY([IdMaterial]) REFERENCES [dbo].[Material]([Id]);

GO
